--klasteris
CREATE CLUSTER TEACHERS_COURSES
(TC_ID NUMBER(4));
--tabulas
CREATE TABLE STUDENTS (
ST_ID NUMBER PRIMARY KEY,
ST_NAME VARCHAR2(30),
ST_SURNAME VARCHAR2(30),
ST_ID_NUM VARCHAR2(9),
ST_GROUP NUMBER)

CREATE TABLE TEACHERS (
T_ID NUMBER(4) PRIMARY KEY,
T_NAME VARCHAR2 (30),
T_TITLE VARCHAR2 (12))
CLUSTER TEACHERS_COURSES (T_ID);

CREATE TABLE COURSES (
C_ID NUMBER(4) PRIMARY KEY,
C_TITLE VARCHAR (30),
C_TEACHER NUMBER,
C_C_POINTS NUMBER CHECK (C_C_POINTS>0), 
CONSTRAINT C_F_K FOREIGN KEY 
(C_TEACHER) REFERENCES TEACHERS (T_ID))
CLUSTER TEACHERS_COURSES (C_ID);

CREATE TABLE GRADES (
G_ID NUMBER PRIMARY KEY,
G_COURSE NUMBER,
G_STUDENT NUMBER,
G_GRADE NUMBER,
G_DATE DATE,
CONSTRAINT F_K_G_C FOREIGN KEY 
(G_COURSE) REFERENCES COURSES(C_ID), 
CONSTRAINT F_K_G_S FOREIGN KEY 
(G_STUDENT) REFERENCES STUDENTS(ST_ID));

-- indeksi
-- bkoka
CREATE INDEX IDX_STUDENTS_ST_ID ON STUDENTS (ST_ID, ST_NAME);
-- bitmap
CREATE BITMAP INDEX IDX_GRADES_G_GRADE 
ON GRADES (STUDENTS.ST_ID, STUDENTS.ST_NAME)
FROM GRADES, STUDENTS 
WHERE GRADES.G_STUDENT = STUDENTS.ST_ID;
-- klasters
CREATE INDEX IDX_CLUSTER_TEACHERS_COURSES ON CLUSTER TEACHERS_COURSES;
--skati
CREATE VIEW VIEWs_STUDENTS_GRADES AS
SELECT 
    S.ST_ID AS Student_ID,
    S.ST_NAME AS Student_Name,
    S.ST_SURNAME AS Student_Surname,
    C.C_TITLE AS Course_Title,
    G.G_GRADE AS Grade,
    G.G_DATE AS Grade_Date
FROM 
    STUDENTS S
JOIN 
    GRADES G ON S.ST_ID = G.G_STUDENT
JOIN 
    COURSES C ON G.G_COURSE = C.C_ID;

CREATE VIEW VIEWs_TEACHERS_COURSES AS
SELECT 
    T.T_ID AS Teacher_ID,
    T.T_NAME AS Teacher_Name,
    T.T_TITLE AS Teacher_Title,
    C.C_TITLE AS Course_Title,
    C.C_C_POINTS AS Course_Credits
FROM 
    TEACHERS T
JOIN 
    COURSES C ON T.T_ID = C.C_TEACHER;
