SELECT
    I1.IELAS_NAME AS "Ielas nosaukums",
    I2.IELAS_NAME AS "Ielas nosaukums"
FROM IELAS I1, IELAS I2
WHERE
    SDO_RELATE(I1.IELAS_GEO, I2.IELAS_GEO,
        'MASK=ANYINTERACT, QUERYTYPE=JOIN') = 'TRUE'
    AND I1.IELAS_ID != I2.IELAS_ID
    AND I1.IELAS_ID > I2.IELAS_ID
ORDER BY I1.IELAS_NAME;

SELECT
    I1.IELAS_ID AS "Ielas 1 ID",
    I2.IELAS_ID AS "Ielas 2 ID",
    SDO_GEOM.SDO_INTERSECTION(I1.IELAS_GEO, I2.IELAS_GEO) AS "Krustošanās ģeometrija"
FROM IELAS I1, IELAS I2
WHERE
    I1.IELAS_ID <> I2.IELAS_ID
    AND I1.IELAS_ID > I2.IELAS_ID
    AND SDO_GEOM.SDO_INTERSECTION(I1.IELAS_GEO, I2.IELAS_GEO) IS NOT NULL;

SELECT
    I.IELAS_ID, I.IELAS_NAME,
    GREATEST(
        SDO_GEOM.SDO_MAX_MBR_ORDINATE(I.IELAS_GEO, 1) - SDO_GEOM.SDO_MIN_MBR_ORDINATE (I.IELAS_GEO, 1),
        SDO_GEOM.SDO_MAX_MBR_ORDINATE(I.IELAS_GEO, 2) - SDO_GEOM.SDO_MIN_MBR_ORDINATE (I.IELAS_GEO, 2)) as "Garums",
    ROUND (SDO_GEOM.SDO_WIDTH (I.IELAS_GEO, 0.05), 2) AS "Platums"
FROM IELAS I;

SELECT
    K.KOKI_ID, K.KOKI_NAME,
    ROUND (SDO_GEOM.SDO_DIAMETER (K.KOKI_GEO, 0.05), 2) AS "Diametrs",
    ROUND (SDO_GEOM.SDO_AREA(K.KOKI_GEO, 0.05), 2) AS "Laukums"
FROM KOKI K;
