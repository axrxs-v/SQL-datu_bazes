CREATE TABLE STUDENTS (
ST_ID NUMBER PRIMARY KEY,
ST_NAME VARCHAR2(30),
ST_SURNAME VARCHAR2(30),
ST_ID_NUM VARCHAR2(9),
ST_GROUP NUMBER)

CREATE TABLE TEACHERS (
T_ID NUMBER(4) PRIMARY KEY,
T_NAME VARCHAR2 (30),
T_TITLE VARCHAR2 (12))

CREATE TABLE COURSES (
C_ID NUMBER(4) PRIMARY KEY,
C_TITLE VARCHAR (30),
C_TEACHER NUMBER,
C_C_POINTS NUMBER CHECK (C_C_POINTS>0), --lielaki par 0
CONSTRAINT C_F_K FOREIGN KEY --saite uz C_TEACHER uz T_ID
(C_TEACHER) REFERENCES TEACHERS (T_ID))

CREATE TABLE GRADES (
G_ID NUMBER PRIMARY KEY,
G_COURSE NUMBER,
G_STUDENT NUMBER,
G_GRADE NUMBER,
G_DATE DATE,
CONSTRAINT F_K_G_C FOREIGN KEY 
(G_COURSE) REFERENCES COURSES(C_ID), --saite C_ID
CONSTRAINT F_K_G_S FOREIGN KEY 
(G_STUDENT) REFERENCES STUDENTS(ST_ID)) --saite ST_ID

FROM COURSES C
JOIN TEACHERS T ON C.C_TEACHER = T.T_ID;

SELECT 
    T.T_NAME AS PASNIEDZĒJS,
    C.C_TITLE AS KURSA_NOSKAUJUMS,
    S.ST_NAME AS STUDENTA_VĀRDS,
    G.G_GRADE AS ATZĪME
FROM 
    GRADES G
JOIN 
    STUDENTS S ON G.G_STUDENT = S.ST_ID
JOIN 
    COURSES C ON G.G_COURSE = C.C_ID
JOIN 
    TEACHERS T ON C.C_TEACHER = T.T_ID
WHERE 
    S.ST_NAME = 'Janis'

    C.C_TITLE;

SELECT 
    s.ST_ID,
    s.ST_NAME,
    s.ST_SURNAME,
    AVG(g.G_GRADE) AS AVG_GRADE_FOR_STUDENT
FROM 
    STUDENTS s
JOIN 
    GRADES g ON s.ST_ID = g.G_STUDENT
GROUP BY 
    s.ST_ID, s.ST_NAME, s.ST_SURNAME
HAVING 
    AVG(g.G_GRADE) >= 7.01;

SELECT DISTINCT 
s.ST_ID, s.ST_NAME, s.ST_SURNAME, s.ST_GROUP
FROM STUDENTS s
JOIN GRADES g ON s.ST_ID = g.G_STUDENT
WHERE g.G_GRADE < 5.00;
