    CONSTRAINT pk_iestades PRIMARY KEY (Iestades_id));
INSERT INTO Iestades_tabula VALUES (Iestades_tips(1, 'IT'));
INSERT INTO Iestades_tabula VALUES (Iestades_tips(2, 'Finansu'));
INSERT INTO Iestades_tabula VALUES (Iestades_tips(3, 'Medicinas'));

CREATE TABLE Apkalpe (
    Apkalpe_id NUMBER PRIMARY KEY,
    Vards VARCHAR2(50),
    Iestades_ref REF Iestades_tips SCOPE IS Iestades_tabula);

DECLARE
    Iestades_ref_IT REF Iestades_tips;
    Iestades_ref_Finansu REF Iestades_tips;
    Iestades_ref_Medicinas REF Iestades_tips;
BEGIN
    SELECT REF(i) INTO Iestades_ref_IT FROM Iestades_tabula i WHERE i.Iestades_id = 1;
    SELECT REF(i) INTO Iestades_ref_Finansu FROM Iestades_tabula i WHERE i.Iestades_id = 2;
    SELECT REF(i) INTO Iestades_ref_Medicinas FROM Iestades_tabula i WHERE i.Iestades_id = 3;

    INSERT INTO Apkalpe VALUES (1, 'Aspazija', Iestades_ref_IT);
    INSERT INTO Apkalpe VALUES (2, 'Krauklis', Iestades_ref_Finansu);
    INSERT INTO Apkalpe VALUES (3, 'Jorens', Iestades_ref_Medicinas);
    INSERT INTO Apkalpe VALUES (4, 'Karlis', Iestades_ref_IT);
END;

SELECT i.Apkalpe_id, i.Vards, DEREF(i.Iestades_ref) AS Iestades
FROM Apkalpe i;

SELECT 
    i.Apkalpe_id, 
    i.Vards, 
    DEREF(i.Iestades_ref).Iestades_nosaukums AS Iesatdes_nos
FROM Apkalpe i;

SELECT 
    i.Apkalpe_id, 
    i.Vards, 
    DEREF(i.Iestades_ref).Iestades_nosaukums AS Iesatdes_nos
FROM Apkalpe i
WHERE DEREF(i.Iestades_ref).Iestades_nosaukums = 'IT';
